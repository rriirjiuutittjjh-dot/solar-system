class SpaceAI { constructor(x, y) { this.x = x; this.y = y; this.vx = 0; this.vy = 0; this.maxSpeed = 6; this.state = "PATROL"; // PATROL, DOCK, or FLEE this.target = null; } // Steering behaviors to navigate toward target planet while compensating for gravity update(planets) { if (!this.target && planets.length > 0) { // Pick nearest planet as target this.target = planets[0]; } if (this.target) { let dx = this.target.x - this.x; let dy = this.target.y - this.y; let dist = Math.hypot(dx, dy); // Seek target planet if (dist > this.target.radius + 20) { let desiredVx = (dx / dist) * this.maxSpeed; let desiredVy = (dy / dist) * this.maxSpeed; // Apply smooth steering force this.vx += (desiredVx - this.vx) * 0.05; this.vy += (desiredVy - this.vy) * 0.05; } else { // Orbit or dock when close this.state = "DOCK"; } } // Apply movement this.x += this.vx; this.y += this.vy; } }
